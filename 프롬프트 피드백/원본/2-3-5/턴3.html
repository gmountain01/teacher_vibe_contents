async function uploadFiles(){
  if(!state.selected)throw new Error('제출함을 선택하세요.');const limits=state.config.limits;
  if(!state.files.length||state.files.length>limits.count||state.files.some(x=>x.file.size>limits.fileMB*1048576)||state.files.reduce((a,x)=>a+x.file.size,0)>limits.totalMB*1048576)throw new Error('파일 수와 용량 제한을 확인하세요.');
  const controls=[...document.querySelectorAll('#app button,#app input')];controls.forEach(b=>b.disabled=true);
  try{
    if(!state.requestId)state.requestId=uid();
    if(!state.receipt)state.receipt=await call('제출시작',Object.assign({},state.parent,{boxId:state.selected,requestId:state.requestId}));
    const known=await call('제출상태',state.receipt);known.files.forEach(f=>{const x=state.files.find(x=>x.id===f.clientId);if(x)x.done=true});
    for(let i=0;i<state.files.length;i++){
      const x=state.files[i];if(x.done)continue;$('uploadProgress').textContent=`파일 ${i+1}/${state.files.length} 저장 중…`;
      try{await call('파일전송',Object.assign({},state.receipt,{clientId:x.id,name:x.file.name,size:x.file.size,mime:x.file.type,base64:await readBase64(x.file)}));x.done=true;x.error=''}catch(e){x.error=e.message}renderFiles();
    }
    const result=await call('제출완료',Object.assign({},state.receipt,{expected:state.files.length}));
    if(result.status==='complete'){
      $('app').innerHTML=`<section class="panel narrow"><div class="steps">3 / 3 · 완료</div><h1>제출이 완료되었습니다</h1><p>${esc((state.boxes.find(b=>b.id===state.selected)||{}).name)}</p><div class="summary">접수번호<br><strong>${esc(result.id)}</strong></div>${state.files.map(x=>`<div class="file">${esc(x.file.name)} <span class="tag">저장 완료</span></div>`).join('')}<p class="muted">제출 확인 시 입력한 임시암호가 필요합니다.</p><div class="actions"><button id="newUpload" class="primary">새 제출</button><button id="checkUpload">업로드 확인</button></div></section>`;

    }else{$('uploadProgress').textContent=`${status(result.status)} · 저장된 파일 ${result.count}개. 실패한 파일을 다시 전송하세요.`}
  }finally{controls.forEach(b=>{if(b.isConnected)b.disabled=false})}
}
